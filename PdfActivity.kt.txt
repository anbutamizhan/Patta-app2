package com.patta.app

import android.app.Activity
import android.graphics.Bitmap
import android.graphics.Color
import android.graphics.pdf.PdfRenderer
import android.os.Bundle
import android.os.CancellationSignal
import android.os.ParcelFileDescriptor
import android.print.*
import android.view.Gravity
import android.widget.*
import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream

class PdfActivity : Activity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val path = intent.getStringExtra("path") ?: return finish()
        val file = File(path)

        val root = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL }
        val bar = LinearLayout(this).apply {
            setBackgroundColor(Color.parseColor("#1565C0"))
            gravity = Gravity.END
            setPadding(16, 8, 16, 8)
        }
        val btn = Button(this).apply { text = "🖨 Print"; setOnClickListener { printFile(file) } }
        bar.addView(btn)
        val list = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL }
        val scroll = ScrollView(this).apply { addView(list) }
        root.addView(bar)
        root.addView(scroll, LinearLayout.LayoutParams(-1, 0, 1f))
        setContentView(root)

        try {
            val w = resources.displayMetrics.widthPixels
            val r = PdfRenderer(ParcelFileDescriptor.open(file, ParcelFileDescriptor.MODE_READ_ONLY))
            for (i in 0 until r.pageCount) {
                val p = r.openPage(i)
                val h = (w.toFloat() / p.width * p.height).toInt()
                val bmp = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)
                bmp.eraseColor(Color.WHITE)
                p.render(bmp, null, null, PdfRenderer.Page.RENDER_MODE_FOR_DISPLAY)
                p.close()
                list.addView(ImageView(this).apply { setImageBitmap(bmp); setPadding(0, 0, 0, 8) })
            }
            r.close()
        } catch (e: Exception) {
            Toast.makeText(this, "PDF காட்ட முடியல: ${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    private fun printFile(file: File) {
        val pm = getSystemService(PRINT_SERVICE) as PrintManager
        pm.print("Patta", object : PrintDocumentAdapter() {
            override fun onLayout(o: PrintAttributes?, n: PrintAttributes?, c: CancellationSignal?,
                                  cb: LayoutResultCallback, e: android.os.Bundle?) {
                cb.onLayoutFinished(PrintDocumentInfo.Builder("patta.pdf")
                    .setContentType(PrintDocumentInfo.CONTENT_TYPE_DOCUMENT).build(), true)
            }
            override fun onWrite(pages: Array<out PageRange>?, d: ParcelFileDescriptor, c: CancellationSignal?, cb: WriteResultCallback) {
                FileInputStream(file).use { i -> FileOutputStream(d.fileDescriptor).use { o -> i.copyTo(o) } }
                cb.onWriteFinished(arrayOf(PageRange.ALL_PAGES))
            }
        }, null)
    }
}
