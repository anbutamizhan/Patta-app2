    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        menu = LinearLayout(this).apply {

        }

            layoutParams = LinearLayout.LayoutParams(-1, dp(110)).apply { setMargins(0, dp(12), 0, dp(12)) }

        webBox = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL; visibility = View.GONE }

        })

            gravity = Gravity.CENTER

        holder.addView(loading)

        root.addView(webBox)

        web.settings.apply {
            javaScriptEnabled = true
            domStorageEnabled = true

    private fun openMode(m: String) {
        mode = m
        menu.visibility = View.GONE
        webBox.visibility = View.VISIBLE
        loading.visibility = View.VISIBLE
        handler.postDelayed({ loading.visibility = View.GONE }, 20000)
        web.loadUrl(SITE_URL)
    }

    private fun showMenu() {
        webBox.visibility = View.GONE
        menu.visibility = View.VISIBLE
        web.loadUrl("about:blank")
    }

    private fun downloadPdf(url: String, ua: String, mime: String?) {
        Toast.makeText(this, "PDF எடுக்கிறது...", Toast.LENGTH_SHORT).show()
        Thread {
            try {
                val c = URL(url).openConnection() as HttpURLConnection
                c.setRequestProperty("User-Agent", ua)
                CookieManager.getInstance().getCookie(url)?.let { c.setRequestProperty("Cookie", it) }
                val f = File(cacheDir, "patta.pdf")
                c.inputStream.use { i -> f.outputStream().use { o -> i.copyTo(o) } }
                handler.post { startActivity(Intent(this, PdfActivity::class.java).putExtra("path", f.absolutePath)) }
            } catch (e: Exception) {
                handler.post { Toast.makeText(this, "PDF எடுக்க முடியல: ${e.message}", Toast.LENGTH_LONG).show() }
            }
        }.start()
    }

    @Deprecated("Deprecated in Java")

        if (webBox.visibility == View.VISIBLE) {
            if (web.canGoBack()) web.goBack() else showMenu()
        } else super.onBackPressed()
    }
